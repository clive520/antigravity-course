function createEcpayOrder(data) { const params = { MerchantID: '2000132', MerchantTradeNo: 'NOVA' + Date.now(), MerchantTradeDate: formatDate(new Date()), PaymentType: 'aio', TotalAmount: data.amount, TradeDesc: 'NovaWave 耳機預購', ItemName: data.plan, ReturnURL: GAS_WEBHOOK_URL, ChoosePayment: 'Credit', EncryptType: '1' }; // 計算 CheckMacValue 檢查碼 params.CheckMacValue = genCheckMac(params); // 產生自動 submit 的 HTML 表單 return renderAutoSubmitForm(params); }
PROMPT ESSENTIALS

提示詞 3 大關鍵要素
